4. Помощники строки 46–77

46// Whole number from min to max, both included
47function randomInt(min, max) {
48  return Math.floor(Math.random() * (max - min + 1)) + min;
49}
50
51function rollCrit() {
52  return Math.random() < CRIT_CHANCE;
53}
54
55function addLog(text) {
56  state.log.unshift(text);              // newest first
57  if (state.log.length > LOG_SIZE) {
58    state.log.length = LOG_SIZE;        // keep only the last N actions
59  }
60}
61
62function setMessage(text, type) {
63  state.message = text;
64  state.messageType = type;
65}
66
67// A rejected move changes nothing except the message
68function reject(text) {
69  setMessage(text, 'error');
70  render();
71}
72
73function getOutcome() {
74  if (state.enemyHp === 0) return 'win';
75  if (state.heroHp === 0) return 'lose';
76  return 'continue';
77}
  • randomInt: целое от min до max включительно
  • rollCrit: Math.random() < 0.1 верно в 10% случаев
  • addLog: unshift кладёт в начало, лишнее обрезаем до LOG_SIZE
  • reject: ставит сообщение об ошибке и рисует, больше ничего не меняет
  • getOutcome: win / lose / continue через ===

5. Действия героя и врага строки 79–111

79function heroAttack() {
80  let damage = randomInt(HERO_MIN_DAMAGE, HERO_MAX_DAMAGE);
81  const crit = rollCrit();
82  if (crit) damage *= CRIT_MULTIPLIER;
83  state.enemyHp = Math.max(0, state.enemyHp - damage);
84  addLog(`You hit the enemy for ${damage}${crit ? ' (CRITICAL HIT!)' : ''}.`);
85}
86
87function heroDefend() {
88  state.defending = true;
89  addLog('You raise your shield: the next hit will be halved.');
90}
91
92function heroHeal() {
93  const before = state.heroHp;
94  state.heroHp = Math.min(MAX_HP, state.heroHp + HEAL_AMOUNT);
95  state.healsLeft -= 1;
96  addLog(`You heal ${state.heroHp - before} HP (${state.healsLeft} heals left).`);
97}
98
99function enemyAttack() {
100  let damage = randomInt(ENEMY_MIN_DAMAGE, ENEMY_MAX_DAMAGE);
101  const crit = rollCrit();
102  if (crit) damage *= CRIT_MULTIPLIER;
103  let note = crit ? ' (CRITICAL HIT!)' : '';
104  if (state.defending) {
105    damage = Math.floor(damage / 2);
106    state.defending = false;            // defence is used up by one hit
107    note += ' (blocked by shield)';
108  }
109  state.heroHp = Math.max(0, state.heroHp - damage);
110  addLog(`The enemy hits you for ${damage}${note}.`);
111}
  • Урон = random; при крите умножаем на 2
  • Math.max(0, ...): HP не уходит ниже нуля
  • В enemyAttack защита делит урон пополам Math.floor(damage/2) и сразу выключается
  • Math.min(MAX_HP, ...) в лечении: выше максимума не лечимся

6. Главная функция takeTurn строки 113–156

113function takeTurn(action) {
114  // --- validation: nothing changes if the move is not allowed ---
115  if (state.over) {
116    reject('The battle is over. Press "New battle" to play again.');
117    return;
118  }
119  if (action !== 'attack' && action !== 'defend' && action !== 'heal') {
120    reject('Unknown move.');
121    return;
122  }
123  if (action === 'defend' && state.defending) {
124    reject('You are already defending. Choose another move.');
125    return;
126  }
127  if (action === 'heal' && state.healsLeft === 0) {
128    reject('No heals left.');
129    return;
130  }
131  if (action === 'heal' && state.heroHp === MAX_HP) {
132    reject('Your HP is already full.');
133    return;
134  }
135
136  // --- hero move ---
137  if (action === 'attack') heroAttack();
138  else if (action === 'defend') heroDefend();
139  else heroHeal();
140
141  // --- enemy answers only if still alive ---
142  state.outcome = getOutcome();
143  if (state.outcome === 'continue') {
144    enemyAttack();
145    state.outcome = getOutcome();
146  }
147
148  state.over = state.outcome === 'win' || state.outcome === 'lose';
149  if (state.over) {
150    setMessage('', '');
151    addLog(state.outcome === 'win' ? 'The enemy is defeated!' : 'You were defeated...');
152  } else {
153    setMessage('', '');
154  }
155  render();
156}
  • Сначала блок проверок: каждая с return, поэтому ход отклоняется до изменений
  • Потом ход героя, затем ответ врага, только если он жив
  • state.over = победа || поражение

8. Рисование (render) строки 171–214

171function renderBar(bar, textEl, hp) {
172  bar.max = MAX_HP;
173  bar.value = hp;
174  bar.classList.toggle('low', (hp / MAX_HP) * 100 <= LOW_HP_PERCENT);
175  textEl.textContent = `${hp} / ${MAX_HP} HP`;
176}
177
178function render() {
179  renderBar(els.heroBar, els.heroHp, state.heroHp);
180  renderBar(els.enemyBar, els.enemyHp, state.enemyHp);
181
182  els.heroDefending.hidden = !state.defending;
183  els.heroCard.classList.toggle('is-defending', state.defending);
184  els.heroCard.classList.toggle('is-dead', state.outcome === 'lose');
185  els.enemyCard.classList.toggle('is-dead', state.outcome === 'win');
186
187  els.message.textContent = state.message;
188  els.message.className = 'message' + (state.messageType ? ' ' + state.messageType : '');
189
190  els.healBtn.textContent = `Heal +${HEAL_AMOUNT} (${state.healsLeft} left)`;
191  for (const btn of els.actions.querySelectorAll('button[data-action]')) {
192    btn.disabled = state.over;
193  }
194
195  els.result.hidden = !state.over;
196  els.result.classList.toggle('win', state.outcome === 'win');
197  els.result.classList.toggle('loss', state.outcome === 'lose');
198  els.result.textContent = state.outcome === 'win'
199    ? 'Victory! You defeated the enemy.'
200    : 'Defeat. The enemy was stronger this time.';
201
202  els.log.replaceChildren();
203  if (state.log.length === 0) {
204    const li = document.createElement('li');
205    li.className = 'empty';
206    li.textContent = 'No actions yet.';
207    els.log.append(li);
208  }
209  for (const entry of state.log) {
210    const li = document.createElement('li');
211    li.textContent = entry;
212    els.log.append(li);
213  }
214}
  • progress.value = hp: полоса обновляется
  • classList.toggle: JS только включает CSS-классы, без inline-стилей
  • btn.disabled = state.over: после конца кнопки неактивны
  • replaceChildren() очищает лог, затем создаём li заново
