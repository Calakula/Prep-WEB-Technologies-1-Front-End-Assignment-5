4. Свои правила строки 99–135

99// ===== Custom rules (what HTML attributes cannot express) =====
100// Each rule returns an error text, or '' when the value is correct
101const rules = {
102  date(value) {
103    const ev = getEvent();
104    if (!ev) return 'Choose an event first.';
105    const slot = ev.dates.find((d) => d.date === value);
106    if (!slot) return `"${ev.name}" is not held on this date. Check the dates under the field.`;
107    if (slot.soldOut) return 'This date is sold out. Choose another date.';
108    return '';
109  },
110  quantity(value) {
111    const type = getType();
112    if (!type) return 'Choose a ticket type first.';
113    if (type.seatsLeft === 0) return `${type.label} tickets are sold out.`;
114    if (Number(value) > type.seatsLeft) return `Only ${type.seatsLeft} ${type.label} seats left.`;
115    return '';
116  },
117  fullName(value) {
118    // at least two words, letters only (hyphen and apostrophe allowed)
119    if (!/^\p{L}[\p{L}'-]*(\s+\p{L}[\p{L}'-]*)+$/u.test(value)) {
120      return 'Enter first and last name using letters only, e.g. "Ravil Maratov".';
121    }
122    return '';
123  },
124  email(value) {
125    const domain = value.split('@')[1] || '';
126    if (BLOCKED_EMAIL_DOMAINS.includes(domain.toLowerCase())) {
127      return 'Temporary email addresses are not accepted. Use your real email.';
128    }
129    return '';
130  },
131  promo(value) {
132    if (value !== '' && !findPromo(value)) return 'Unknown promo code.';
133    return '';
134  },
135};
  • Каждое правило возвращает текст ошибки или пустую строку
  • Имя: регулярка \p{L} (любые буквы) и ≥2 слова
  • Почта: домен не из BLOCKED_EMAIL_DOMAINS

5. Свои тексты вместо браузерных строки 137–149

137// Own messages instead of the browser ones
138function nativeMessage(field, label) {
139  const v = field.validity;
140  if (v.valueMissing) return `${label} is required.`;
141  if (v.typeMismatch) return 'Enter a valid email address, e.g. name@example.com.';
142  if (v.tooShort) return `${label} must have at least ${field.minLength} characters.`;
143  if (v.tooLong) return `${label} must have at most ${field.maxLength} characters.`;
144  if (v.rangeUnderflow) return `Minimum value is ${field.min}.`;
145  if (v.rangeOverflow) return `You can buy at most ${field.max} tickets per order.`;
146  if (v.stepMismatch || v.badInput) return 'Enter a whole number.';
147  if (v.patternMismatch) return `${label} may contain only letters and digits.`;
148  return 'Invalid value.';
149}
  • field.validity: флаги valueMissing, typeMismatch, rangeOverflow и др.

6. validateField строки 151–182

151// ===== Validation =====
152function validateField(name, show = true) {
153  const field = fields[name];
154  const value = field.value.trim();
155  field.setCustomValidity('');          // clear the old custom message first
156  let message = '';
157  if (!field.validity.valid) {
158    message = nativeMessage(field, LABELS[name]);
159  } else if (rules[name]) {
160    message = rules[name](value);
161  }
162  if (message) field.setCustomValidity(message); // checkValidity() now agrees with us
163  if (show) showFieldState(name, message);
164  return message;
165}
166
167function showFieldState(name, message) {
168  const field = fields[name];
169  errorEls[name].textContent = message;
170  field.setAttribute('aria-invalid', message ? 'true' : 'false');
171  field.classList.toggle('is-invalid', message !== '');
172  field.classList.toggle('is-valid', message === '' && field.value.trim() !== '');
173}
174
175function clearFieldState(name) {
176  const field = fields[name];
177  delete field.dataset.touched;
178  field.removeAttribute('aria-invalid');
179  field.classList.remove('is-invalid', 'is-valid');
180  field.setCustomValidity('');
181  errorEls[name].textContent = '';
182}
  • Сначала setCustomValidity(""): сброс старой ошибки
  • Если браузерная проверка не пройдена → её текст, иначе наше правило
  • aria-invalid и классы is-invalid/is-valid показывают результат

185function addRow(list, label, value) {
186  const dt = document.createElement('dt');
187  dt.textContent = label;

189  dd.textContent = value;

193function getTotals() {
194  const type = getType();
195  const quantity = Number(fields.quantity.value);
196  const promo = findPromo(fields.promo.value);
197  if (!type || !(quantity > 0)) return null;
198  const subtotal = type.price * quantity;
199  const discount = promo ? promo.discount : 0;
200  return { subtotal, discount, total: Math.round(subtotal * (100 - discount) / 100) };
201}
202
203function renderPreview() {

205  const type = getType();
206  const totals = getTotals();

209  addRow(list, 'Date', fields.date.value || '—');
210  addRow(list, 'Type', type ? type.label : '—');
211  addRow(list, 'Quantity', fields.quantity.value || '—');
212  addRow(list, 'Name', fields.fullName.value.trim() || '—');
213  addRow(list, 'Email', fields.email.value.trim() || '—');
214  addRow(list, 'Discount', totals && totals.discount > 0 ? `-${totals.discount}%` : '—');

8. События формы строки 219–267

219// ===== Events =====
220for (const name of FIELD_NAMES) {
221  const field = fields[name];
222
223  // blur: the field becomes "touched" and is checked for the first time
224  field.addEventListener('blur', () => {
225    field.dataset.touched = 'true';
226    validateField(name);
227  });
228
229  // input: preview always updates; the check runs only after the first blur
230  field.addEventListener('input', () => {
231    if (name === 'event') updateDateHint();
232    if (field.dataset.touched === 'true') validateField(name);
233    for (const dep of DEPENDENTS[name] || []) {
234      if (fields[dep].dataset.touched === 'true') validateField(dep);
235    }
236    renderPreview();
237  });
238}
239
240form.addEventListener('submit', (event) => {
241  event.preventDefault();
242  const invalid = [];
243  for (const name of FIELD_NAMES) {
244    fields[name].dataset.touched = 'true';
245    if (validateField(name)) invalid.push(name);
246  }
247
248  if (invalid.length > 0) {
249    summary.textContent = `The form has ${invalid.length} error${invalid.length === 1 ? '' : 's'}. Please fix the marked fields.`;
250    summary.hidden = false;
251    fields[invalid[0]].focus();
252    return;
253  }
254
255  summary.hidden = true;
256  createTicket();
257});
258
259form.addEventListener('reset', () => {
260  // wait until the browser has cleared the values
261  setTimeout(() => {
262    for (const name of FIELD_NAMES) clearFieldState(name);
263    summary.hidden = true;
264    updateDateHint();
265    renderPreview();
266  }, 0);
267});
  • blur: помечаем поле «тронутым» и проверяем
  • input: предпросмотр всегда, проверка только если поле тронуто, плюс зависимые поля
  • submit: preventDefault(), проверяем все; при ошибках счётчик и focus() на первом
  • reset: чистим состояния полей

269// ===== Final result =====
270function createBookingCode() {
271  let code = '';
272  for (let i = 0; i < BOOKING_LENGTH; i += 1) {
273    code += BOOKING_ALPHABET[randomInt(0, BOOKING_ALPHABET.length - 1)];
274  }
275  return `${BOOKING_PREFIX}-${code}`;
276}
277
278function createTicket() {
279  const ev = getEvent();
280  const type = getType();
281  const quantity = Number(fields.quantity.value);
282  const totals = getTotals();
283  const data = {
284    event: ev.name, date: fields.date.value, type: type.label, quantity,
285    name: fields.fullName.value.trim(), email: fields.email.value.trim(),
286    discount: totals.discount, total: totals.total,
287  };
288
289  type.seatsLeft -= quantity; // the seats are really taken
290
291  const ticket = document.createElement('article');
292  ticket.className = 'ticket final';
293  ticket.tabIndex = -1;
294  const title = document.createElement('h2');
295  title.textContent = 'Booking confirmed';
296  const list = document.createElement('dl');
297  addRow(list, 'Event', data.event);
298  addRow(list, 'Date', data.date);
299  addRow(list, 'Tickets', `${data.quantity} × ${data.type}`);
300  addRow(list, 'Name', data.name);
301  addRow(list, 'Email', data.email);
302  addRow(list, 'Paid', formatMoney(data.total) + (data.discount > 0 ? ` (-${data.discount}%)` : ''));
303  const code = document.createElement('p');
304  code.className = 'code';
305  code.textContent = createBookingCode();
306  ticket.append(title, list, code);
307  result.replaceChildren(ticket);
308
309  form.reset();           // the reset listener clears states and the preview
310  buildTypeOptions();     // show the new seats left
311  ticket.focus();
312}
  • Код брони: randomInt по алфавиту ×6
  • seatsLeft -= quantity: места реально занимаются
  • Билет строится через DOM; form.reset() и перестройка опций
