3. Свои циклы строки 40–63

40// ===== Calculations with my own loops =====
41function countByStatus(status) {
42  let count = 0;
43  for (const card of cards) {
44    if (card.status === status) count += 1;
45  }
46  return count;
47}
48
49function hoursLeft() {
50  let sum = 0;
51  for (const card of cards) {
52    if (card.status !== 'done') sum += card.hours;
53  }
54  return sum;
55}
56
57function getTags() {
58  const tags = [];
59  for (const card of cards) {
60    if (!tags.includes(card.tag)) tags.push(card.tag);
61  }
62  return tags.sort();
63}
  • countByStatus: счётчик в цикле for…of
  • hoursLeft: сумма часов невыполненных
  • getTags: уникальные теги через includes

4. Фильтр и сортировка строки 65–78

65// ===== Filtering and sorting (returns a new array, data is not changed) =====
66function getVisibleCards() {
67  const query = state.query.trim().toLowerCase();
68  const result = [];
69  for (const card of cards) {
70    const matchesTag = state.tag === 'all' || card.tag === state.tag;
71    const matchesQuery = query === '' || card.title.toLowerCase().includes(query);
72    if (matchesTag && matchesQuery) result.push(card);
73  }
74  if (state.sort === 'high') result.sort((a, b) => b.priority - a.priority || a.id - b.id);
75  else if (state.sort === 'low') result.sort((a, b) => a.priority - b.priority || a.id - b.id);
76  else result.sort((a, b) => a.id - b.id);
77  return result;
78}
  • Собираем новый массив result: оригинал не трогаем
  • &&: карточка нужна, если подходит и тег, и текст
  • sort((a,b)=>b.priority-a.priority): от большего к меньшему

5. Действие moveCard строки 80–105

80// ===== Actions (change the array first, then render) =====
81function setMessage(text, type) {
82  state.message = text;
83  state.messageType = type;
84}
85
86function moveCard(id, direction) {
87  const card = cards.find((c) => c.id === id);
88  if (!card) return;
89  const target = STATUSES[STATUSES.indexOf(card.status) + direction];
90  if (!target) return;
91
92  // Rule: "In progress" holds at most WIP_LIMIT cards
93  if (target === 'doing' && countByStatus('doing') >= WIP_LIMIT) {
94    setMessage(
95      `Cannot move "${card.title}" to ${STATUS_TITLES.doing}: the column already has ${WIP_LIMIT} cards. Finish one first.`,
96      'error'
97    );
98    render();
99    return;
100  }
101
102  card.status = target;
103  setMessage(`Moved "${card.title}" to ${STATUS_TITLES[target]}.`, 'ok');
104  render();
105}
  • find ищет карточку по id
  • STATUSES.indexOf(...)+direction: следующая или прошлая колонка
  • Правило лимита → сообщение, render(), return
  • Иначе меняем card.status и зовём render()

6. Создание карточки строки 107–156

107// ===== Rendering =====
108function createButton(text, action, card, label, disabled) {
109  const btn = document.createElement('button');
110  btn.type = 'button';
111  btn.className = 'btn btn-secondary btn-small';
112  btn.textContent = text;
113  btn.dataset.action = action;
114  btn.dataset.id = String(card.id);
115  btn.setAttribute('aria-label', label);
116  btn.disabled = disabled;
117  return btn;
118}
119
120function createCard(card) {
121  const li = document.createElement('li');
122  li.className = `card priority-${card.priority}`;
123  li.classList.toggle('is-done', card.status === 'done');
124
125  const title = document.createElement('h3');
126  title.textContent = card.title;
127
128  const meta = document.createElement('p');
129  meta.className = 'meta';
130  const parts = [`#${card.tag}`, `${PRIORITY_LABELS[card.priority]} priority`, `${card.hours} h`];
131  for (const text of parts) {
132    const chip = document.createElement('span');
133    chip.className = 'chip';
134    chip.textContent = text;
135    meta.append(chip);
136  }
137  if (card.urgent) {
138    const chip = document.createElement('span');
139    chip.className = 'chip urgent';
140    chip.textContent = 'Urgent';
141    meta.append(chip);
142  }
143
144  const index = STATUSES.indexOf(card.status);
145  const actions = document.createElement('div');
146  actions.className = 'card-actions';
147  const prev = STATUSES[index - 1];
148  const next = STATUSES[index + 1];
149  actions.append(
150    createButton('← Back', 'back', card, prev ? `Move "${card.title}" to ${STATUS_TITLES[prev]}` : 'Cannot go back', !prev),
151    createButton('Next →', 'next', card, next ? `Move "${card.title}" to ${STATUS_TITLES[next]}` : 'Already done', !next)
152  );
153
154  li.append(title, meta, actions);
155  return li;
156}
  • createElement + textContent: безопасно, без innerHTML
  • dataset.action, dataset.id: метки для делегирования
  • disabled у «Back» в первой колонке и «Next» в последней

7. Рисование всей страницы строки 158–227

158function renderFilterOptions() {
159  const tags = getTags();
160  if (state.tag !== 'all' && !tags.includes(state.tag)) state.tag = 'all';
161  els.tag.replaceChildren();
162  for (const value of ['all', ...tags]) {
163    const option = document.createElement('option');
164    option.value = value;
165    option.textContent = value === 'all' ? 'All tags' : value;
166    els.tag.append(option);
167  }
168  els.tag.value = state.tag;
169}
170
171function renderProgress() {
172  const done = countByStatus('done');
173  const percent = cards.length === 0 ? 0 : Math.round((done / cards.length) * 100);
174  els.progress.max = Math.max(cards.length, 1);
175  els.progress.value = done;
176  els.progressText.textContent = `${done} of ${cards.length} cards finished (${percent}%). Hours left: ${hoursLeft()}.`;
177}
178
179function renderBoard(visible) {
180  els.board.replaceChildren();
181  for (const status of STATUSES) {
182    const column = document.createElement('section');
183    column.className = 'column';
184    const total = countByStatus(status);
185    const isLimited = status === 'doing';
186    column.classList.toggle('is-full', isLimited && total >= WIP_LIMIT);
187
188    const heading = document.createElement('h2');
189    const name = document.createElement('span');
190    name.textContent = STATUS_TITLES[status];
191    const count = document.createElement('span');
192    count.textContent = isLimited ? `${total}/${WIP_LIMIT}` : String(total);
193    heading.append(name, count);
194
195    const list = document.createElement('ul');
196    list.className = 'card-list';
197    let shown = 0;
198    for (const card of visible) {
199      if (card.status !== status) continue;
200      list.append(createCard(card));
201      shown += 1;
202    }
203    if (shown === 0) {
204      const empty = document.createElement('li');
205      empty.className = 'empty-note';
206      empty.textContent = 'No cards here';
207      list.append(empty);
208    }
209    column.append(heading, list);
210    els.board.append(column);
211  }
212}
213
214function render() {
215  const visible = getVisibleCards();
216  renderFilterOptions();
217  renderProgress();
218  renderBoard(visible);
219
220  els.message.textContent = state.message;
221  els.message.className = 'message' + (state.messageType ? ' ' + state.messageType : '');
222
223  els.emptyBoard.hidden = visible.length > 0;
224  if (visible.length === 0) {
225    els.emptyBoard.textContent = 'Nothing found. Try another word or reset the filters.';
226  }
227}
  • renderBoard: для каждого статуса строим колонку и кладём карточки
  • Если карточек нет: «No cards here»
  • render(): фильтры → прогресс → доска → сообщение; «Nothing found», если ничего не подошло
